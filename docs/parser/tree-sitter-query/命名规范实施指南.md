use serde::{Deserialize, Serialize};
use std::fmt;

/// 查询捕获名称的域
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub enum Domain {
    #[serde(rename = "entity")]
    Entity,
    #[serde(rename = "call")]
    Call,
    #[serde(rename = "dependency")]
    Dependency,
    #[serde(rename = "comment")]
    Comment,
}

impl fmt::Display for Domain {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Domain::Entity => write!(f, "entity"),
            Domain::Call => write!(f, "call"),
            Domain::Dependency => write!(f, "dependency"),
            Domain::Comment => write!(f, "comment"),
        }
    }
}

impl std::str::FromStr for Domain {
    type Err = String;

    fn from_str(s: &str) -> Result<Self, Self::Err> {
        match s {
            "entity" => Ok(Domain::Entity),
            "call" => Ok(Domain::Call),
            "dependency" => Ok(Domain::Dependency),
            "comment" => Ok(Domain::Comment),
            _ => Err(format!("Unknown domain: {}", s)),
        }
    }
}

/// 捕获名称的解析结构
#[derive(Debug, Clone, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub struct CaptureName {
    pub domain: Domain,
    pub category: Option<String>,
    pub subtype: Option<String>,
    pub role: Option<String>,
    pub attribute: Option<String>,
}

impl CaptureName {
    /// 解析捕获名称字符串
    ///
    /// # 示例
    ///
    /// ```
    /// let capture = CaptureName::parse("@entity.type.class.name").unwrap();
    /// assert_eq!(capture.domain, Domain::Entity);
    /// assert_eq!(capture.category, Some("type".to_string()));
    /// assert_eq!(capture.subtype, Some("class".to_string()));
    /// assert_eq!(capture.role, Some("name".to_string()));
    /// assert_eq!(capture.attribute, None);
    /// ```
    pub fn parse(capture: &str) -> Result<Self, ParseError> {
        // 移除 @ 前缀
        let capture = capture.strip_prefix('@').unwrap_or(capture);

        // 分割各部分
        let parts: Vec<&str> = capture.split('.').collect();

        if parts.len() < 2 {
            return Err(ParseError::InvalidFormat {
                capture: capture.to_string(),
                reason: "Must have at least domain and category".to_string(),
            });
        }

        // 解析 domain
        let domain = Domain::from_str(parts[0]).map_err(|_| ParseError::UnknownDomain {
            domain: parts[0].to_string(),
        })?;

        // 解析其他部分
        Ok(CaptureName {
            domain,
            category: if parts.len() > 1 { Some(parts[1].to_string()) } else { None },
            subtype: if parts.len() > 2 { Some(parts[2].to_string()) } else { None },
            role: if parts.len() > 3 { Some(parts[3].to_string()) } else { None },
            attribute: if parts.len() > 4 { Some(parts[4].to_string()) } else { None },
        })
    }

    /// 获取完整的捕获名称字符串
    pub fn to_string(&self) -> String {
        let mut parts = vec![self.domain.to_string()];

        if let Some(ref cat) = self.category {
            parts.push(cat.clone());
        }
        if let Some(ref sub) = self.subtype {
            parts.push(sub.clone());
        }
        if let Some(ref role) = self.role {
            parts.push(role.clone());
        }
        if let Some(ref attr) = self.attribute {
            parts.push(attr.clone());
        }

        format!("@{}", parts.join("."))
    }

    /// 检查是否匹配给定的模式
    ///
    /// # 示例
    ///
    /// ```
    /// let capture = CaptureName::parse("@entity.type.class.name").unwrap();
    /// assert!(capture.matches(Domain::Entity, Some("type"), Some("class"), Some("name"), None));
    /// ```
    pub fn matches(
        &self,
        domain: Domain,
        category: Option<&str>,
        subtype: Option<&str>,
        role: Option<&str>,
        attribute: Option<&str>,
    ) -> bool {
        if self.domain != domain {
            return false;
        }

        if let Some(cat) = category {
            if self.category.as_deref() != Some(cat) {
                return false;
            }
        }

        if let Some(sub) = subtype {
            if self.subtype.as_deref() != Some(sub) {
                return false;
            }
        }

        if let Some(r) = role {
            if self.role.as_deref() != Some(r) {
                return false;
            }
        }

        if let Some(attr) = attribute {
            if self.attribute.as_deref() != Some(attr) {
                return false;
            }
        }

        true
    }
}

impl fmt::Display for CaptureName {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(f, "{}", self.to_string())
    }
}

/// 解析错误类型
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    InvalidFormat { capture: String, reason: String },
    UnknownDomain { domain: String },
}

impl fmt::Display for ParseError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            ParseError::InvalidFormat { capture, reason } => {
                write!(f, "Invalid capture format '{}': {}", capture, reason)
            }
            ParseError::UnknownDomain { domain } => {
                write!(f, "Unknown domain: {}", domain)
            }
        }
    }
}

impl std::error::Error for ParseError {}
```

    CaptureName::parse(capture)
        .map(|c| c.domain == Domain::Call)

/// 检查捕获名称是否属于依赖关系
pub fn is_dependency_capture(capture: &str) -> bool {
    CaptureName::parse(capture)
        .map(|c| c.domain == Domain::Dependency)
        .unwrap_or(false)
}

/// 提取实体类型
pub fn extract_entity_type(capture: &str) -> Option<String> {
    CaptureName::parse(capture)
        .ok()
        .filter(|c| c.domain == Domain::Entity)
        .and_then(|c| c.category)
}

/// 提取调用类型
pub fn extract_call_type(capture: &str) -> Option<String> {
    CaptureName::parse(capture)
        .ok()
        .filter(|c| c.domain == Domain::Call)
        .and_then(|c| c.category)
}

/// 验证查询中的所有捕获名称是否符合规范
pub fn validate_capture_names(query: &str, language: &str) -> Result<Vec<String>, ValidationError> {
    let mut errors = Vec::new();
    let mut valid_captures = Vec::new();

    // 提取所有 @ 开头的捕获名称
    for line in query.lines() {
        let captures: Vec<&str> = line
            .split('@')
            .skip(1)
            .filter_map(|s| s.split_whitespace().next())
            .collect();

        for capture in captures {
            match CaptureName::parse(&format!("@{}", capture)) {
                Ok(_) => valid_captures.push(capture.to_string()),
                Err(e) => errors.push(ValidationError {
                    capture: capture.to_string(),
                    error: e.to_string(),
                    language: language.to_string(),
                }),
            }
        }
    }

    if errors.is_empty() {
        Ok(valid_captures)
    } else {
        Err(errors)
    }
}

/// 验证错误
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ValidationError {
    pub capture: String,
    pub error: String,
    pub language: String,
}

impl fmt::Display for ValidationError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(
            f,
            "Validation error in {}: capture '{}': {}",
            self.language, self.capture, self.error
        )
    }
}

### 迁移脚本

创建迁移脚本 `src/query/scheme/migrate.rs`：

```rust
use std::collections::HashMap;

/// 旧命名到新命名的映射
pub fn get_migration_map() -> HashMap<&'static str, &'static str> {
    let mut map = HashMap::new();

    // C 语言迁移映射
    map.insert("@entity.variable.normal.name", "@entity.variable.declaration.name");
    map.insert("@entity.variable.normal.type", "@entity.variable.declaration.type");
    map.insert("@entity.variable.normal.value", "@entity.variable.declaration.value");
    map.insert("@entity.component.field.name", "@entity.variable.field.name");
    map.insert("@entity.component.field.type", "@entity.variable.field.type");
    map.insert("@entity.component.parameter.name", "@entity.variable.parameter.name");
    map.insert("@entity.component.parameter.type", "@entity.variable.parameter.type");

    // C# 语言迁移映射
    map.insert("@call.method.object.name", "@call.method.instance.object");
    map.insert("@call.method.function.name", "@call.method.instance.function");
    map.insert("@call.static.class.name", "@call.method.static.class");
    map.insert("@call.static.method.name", "@call.method.static.function");

    // C++ 语言迁移映射（继承自 C）
    map.extend(get_migration_map()); // 使用 C 语言的映射

    map
}

/// 迁移查询字符串中的捕获名称
pub fn migrate_query(query: &str) -> String {
    let migration_map = get_migration_map();
    let mut result = query.to_string();

    for (old_name, new_name) in migration_map {
        result = result.replace(old_name, new_name);
    }

    result
}

        assert_eq!(capture.subtype, Some("instance".to_string()));
        assert_eq!(capture.role, Some("object".to_string()));
