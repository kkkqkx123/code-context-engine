## 二、摘要生成策略

### 策略1：结构化摘要（推荐）

不依赖模型"理解"，而是引导模型提取结构化信息：

```python
PROMPT_TEMPLATE = """Analyze this code file and generate a structured summary:

{code_content}

Output format (strictly follow this structure):
1. FILE_PURPOSE: One sentence describing the main purpose (max 20 words)
2. KEY_COMPONENTS: List main classes/functions (max 5 items)
3. DEPENDENCIES: Key external imports/libraries used
4. CORE_LOGIC: Brief description of the main workflow (2-3 sentences)
5. TAGS: 3-5 keywords for indexing

Example:
1. FILE_PURPOSE: User authentication and session management
2. KEY_COMPONENTS: AuthController, UserService, JwtUtils
3. DEPENDENCIES: flask, jwt, bcrypt
4. CORE_LOGIC: Handles login requests, validates credentials, generates JWT tokens with 24h expiry
5. TAGS: authentication, jwt, security, session

Now generate for the code above:"""

def generate_structured_summary(code_content, model):
    """生成结构化摘要，便于后续检索"""
    response = model.generate(PROMPT_TEMPLATE.format(code_content=code_content))
    return parse_structured_response(response)
```

### 策略2：分层摘要生成[针对较长的文件]

避免让小模型处理整个大文件，而是先聚合再总结：

```python
def generate_hierarchical_summary(file_path, ast_entities, model, max_tokens=4096):
    """
    分层摘要：先摘要各个组件，再合并成文件摘要
    """
    # 第1层：组件级摘要
    component_summaries = []

    # 对每个类/函数生成简短摘要
    for entity in ast_entities:
        if entity.type in ['class', 'function']:
            component_summary = generate_component_summary(
                entity, model, max_tokens=512
            )
            component_summaries.append({
                "name": entity.name,
                "type": entity.type,
                "summary": component_summary,
                "lines": f"{entity.start_line}-{entity.end_line}"
            })

    # 第2层：文件级摘要（基于组件摘要聚合）
    if len(code_content) > max_tokens * 3:
        # 文件太大，不传完整代码，只传组件摘要
        file_summary = generate_file_summary_from_components(
            file_path, component_summaries, model
        )
    else:
        # 文件适中，用完整代码生成
        file_summary = generate_full_file_summary(
            file_path, code_content, model
        )

    return {
        "file_summary": file_summary,
        "component_summaries": component_summaries
    }
```

## 三、工程优化实践

### 1. **缓存与增量更新**

```python
class SummaryCache:
    """避免重复生成摘要"""

    def __init__(self, cache_dir="./summary_cache"):
        self.cache_dir = cache_dir
        self.summary_hash = {}  # file_path -> content_hash

    def get_or_generate(self, file_path, content, model):
        # 计算文件哈希
        content_hash = hashlib.md5(content.encode()).hexdigest()

        # 检查缓存
        cache_file = f"{self.cache_dir}/{file_path.replace('/', '_')}.json"
        if os.path.exists(cache_file):
            with open(cache_file) as f:
                cached = json.load(f)
                if cached.get("hash") == content_hash:
                    return cached["summary"]

        # 生成新摘要
        summary = self.generate_summary(content, model)

        # 保存缓存
        with open(cache_file, 'w') as f:
            json.dump({
                "hash": content_hash,
                "summary": summary,
                "timestamp": time.time()
            }, f)

        return summary
```

### 2. **批处理与异步队列**

```python
import asyncio
from concurrent.futures import ThreadPoolExecutor

class BatchSummarizer:
    """批量生成摘要，避免频繁模型调用"""

    def __init__(self, model, batch_size=10):
        self.model = model
        self.batch_size = batch_size
        self.queue = asyncio.Queue()

    async def batch_generate(self, files):
        """批量生成摘要"""
        # 分批处理
        for i in range(0, len(files), self.batch_size):
            batch = files[i:i+self.batch_size]
            tasks = [self.generate_async(file) for file in batch]
            results = await asyncio.gather(*tasks)
            yield results

    async def generate_async(self, file):
        """异步生成单个摘要"""
        loop = asyncio.get_event_loop()
        with ThreadPoolExecutor() as pool:
            result = await loop.run_in_executor(
                pool,
                self.model.generate,
                self.build_prompt(file)
            )
        return result
```

### 3. **降级策略与兜底**

当模型不可用或速度太慢时，使用基于规则的摘要：

```python
def fallback_summary(code_content, ast_entities):
    """
    基于规则的降级摘要（不依赖模型）
    """
    summary = []

    # 1. 提取导入的框架
    imports = extract_imports(ast_entities)
    main_framework = detect_main_framework(imports)
    if main_framework:
        summary.append(f"Uses {main_framework}")

    # 2. 提取主要组件
    classes = [e for e in ast_entities if e.type == 'class']
    functions = [e for e in ast_entities if e.type == 'function']

    if classes:
        summary.append(f"Defines {len(classes)} classes: {', '.join(c.name for c in classes[:3])}")
    if functions:
        summary.append(f"Defines {len(functions)} functions: {', '.join(f.name for f in functions[:3])}")

    # 3. 从docstring提取描述
    main_class = classes[0] if classes else None
    if main_class and main_class.docstring:
        summary.append(f"Main purpose: {main_class.docstring[:100]}")

    # 4. 代码特征
    if has_decorator(ast_entities, 'route') or has_decorator(ast_entities, 'app.'):
        summary.append("Contains web route definitions")
    if has_import(imports, 'sqlalchemy') or has_import(imports, 'django.db'):
        summary.append("Database ORM operations")

    return " | ".join(summary) if summary else "Code file with multiple components"
```

def should_generate_summary(file_stats):
    """判断是否值得生成摘要"""

    # 跳过过小的文件
    if file_stats["lines"] < 10:
        return False, "too_small"

    # 跳过纯配置文件

    # 代码文件都生成摘要

## 五、推荐方案总结

### 混合策略（兼顾质量与成本）

```python
class HybridSummaryGenerator:
    """
    80%文件用规则生成，20%核心文件用模型生成
    """

    def __init__(self, model):
        self.model = model
        self.rule_summarizer = RuleBasedSummarizer()

    def generate(self, file_path, content, ast_entities, importance_score):
        """
        importance_score: 基于文件大小、调用次数、修改频率等计算
        """
        # 高重要性文件使用模型摘要
        if importance_score > 0.7:
            return self.model_summary(content)

        # 中等文件：混合策略
        elif importance_score > 0.3:
            rule_summary = self.rule_summarizer.generate(content, ast_entities)
            # 如果规则摘要足够好（长度>50字符），就用规则的
            if len(rule_summary) > 50:
                return rule_summary
            else:
                return self.model_summary(content)

        # 低重要性：纯规则摘要
        else:
            return self.rule_summarizer.generate(content, ast_entities)
```

**实际落地建议**：

1. **初期**：用规则摘要 + 少量模型验证，观察效果
2. **中期**：对核心模块使用小模型，其他用规则
3. **成熟期**：建立缓存机制，只对变更文件重新生成

8B小模型完全够用，关键是**控制输入长度**（通过AST提取关键部分）和**结构化输出**（便于检索），而不是让模型理解整个大文件。
