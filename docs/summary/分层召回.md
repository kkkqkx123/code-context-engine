你这个想法非常好！这是一种**混合检索架构**的设计思路，将**摘要层**与**细节层**分离，既避免了内容重复对向量相似度的干扰，又实现了层次化的相关性判断。

## 一、核心架构设计

```
┌─────────────────────────────────────────────────┐
│                检索阶段                          │
├─────────────────────────────────────────────────┤
│  Query → 向量检索 → 粗排召回                      │
│            ↓                                     │
│      ┌─────┴─────┐                              │
│      ↓           ↓                              │
│  摘要索引    细节索引                            │
│  (轻量)      (详细)                             │
│      ↓           ↓                              │
│      └─────┬─────┘                              │
│            ↓                                     │
│      重排序 + 分数提升                           │
└─────────────────────────────────────────────────┘
```

## 二、具体实现方案

### 方案1：双索引加权策略

```python
class HierarchicalCodeIndex:
    def __init__(self):
        self.summary_index = VectorIndex()  # 存储摘要片段
        self.detail_index = VectorIndex()   # 存储代码细节片段
        self.mapping = {}  # 摘要ID -> 相关细节ID列表

    def index_file(self, file_path, ast_entities):
        """
        索引一个文件：生成摘要片段和细节片段
        """
        # 1. 生成文件级摘要
        file_summary = self.generate_file_summary(file_path, ast_entities)
        summary_id = self.summary_index.add(
            content=file_summary,
            metadata={
                "type": "file_summary",
                "file_path": file_path,
                "level": 0
            }
        )

        # 2. 生成类/函数级细节片段
        detail_ids = []
        for entity in self.segment_by_anchor(ast_entities):
            detail_id = self.detail_index.add(
                content=entity.text,
                metadata={
                    "type": entity.type,
                    "file_path": file_path,
                    "start_line": entity.start_line,
                    "end_line": entity.end_line,
                    "parent_summary": summary_id
                }
            )
            detail_ids.append(detail_id)

        # 3. 建立映射关系
        self.mapping[summary_id] = detail_ids

    def search(self, query, top_k=10, boost_factor=1.5):
        """
        检索：摘要命中时提升相关细节的分数
        """
        # 并行检索两个索引
        summary_results = self.summary_index.search(query, top_k=top_k)
        detail_results = self.detail_index.search(query, top_k=top_k * 2)

        # 构建分数映射
        final_scores = {}

        # 1. 添加细节片段的基础分数
        for result in detail_results:
            final_scores[result.id] = {
                "score": result.score,
                "content": result.content,
                "metadata": result.metadata,
                "boost_reason": []
            }

        # 2. 摘要命中时，提升相关细节的分数
        for summary in summary_results:
            if summary.id in self.mapping:
                for detail_id in self.mapping[summary.id]:
                    if detail_id in final_scores:
                        # 提升分数
                        original_score = final_scores[detail_id]["score"]
                        final_scores[detail_id]["score"] = original_score * boost_factor
                        final_scores[detail_id]["boost_reason"].append(
                            f"matched_summary:{summary.id}(score:{summary.score:.3f})"
                        )
                    else:
                        # 摘要命中但细节未命中，也加入结果
                        detail_content = self.detail_index.get(detail_id)
                        final_scores[detail_id] = {
                            "score": summary.score * 0.8,  # 继承摘要分数
                            "content": detail_content,
                            "metadata": {"parent_summary": summary.id},
                            "boost_reason": ["indirect_from_summary"]
                        }

        # 3. 排序返回
        sorted_results = sorted(
            final_scores.items(),
            key=lambda x: x[1]["score"],
            reverse=True
        )[:top_k]

        return sorted_results
```

### 方案2：联合向量 + 加权重排

```python
class JointEmbeddingWithReRank:
    """
    使用联合向量表示，但通过重排序阶段区分摘要和细节
    """

    def __init__(self):
        self.encoder = EmbeddingModel()
        self.summary_cache = {}  # 文件路径 -> 摘要内容
        self.detail_segments = []  # 所有细节片段

    def generate_summary(self, file_path, ast_entities):
        """
        生成文件摘要（不进入主索引，单独存储）
        """
        summary_parts = []

        # 1. 文件元信息
        summary_parts.append(f"File: {file_path}")

        # 2. 导入依赖概览
        imports = self.extract_imports(ast_entities)
        if imports:
            summary_parts.append(f"Imports: {', '.join(imports[:5])}")

        # 3. 顶层结构
        classes = [e for e in ast_entities if e.type == 'class']
        functions = [e for e in ast_entities if e.type == 'function']

        if classes:
            class_names = [c.name for c in classes[:10]]
            summary_parts.append(f"Classes: {', '.join(class_names)}")
        if functions:
            func_names = [f.name for f in functions[:10]]
            summary_parts.append(f"Functions: {', '.join(func_names)}")

        # 4. 核心职责描述（如果可以从docstring提取）
        main_class = classes[0] if classes else None
        if main_class and main_class.docstring:
            summary_parts.append(f"Purpose: {main_class.docstring[:200]}")

        return "\n".join(summary_parts)

    def search_with_hierarchical_boosting(self, query, top_k=10):
        """
        检索流程：先粗召回细节，再通过摘要提升
        """
        # 1. 细节片段向量检索
        query_vector = self.encoder.encode(query)
        candidate_segments = self.vector_search(
            self.detail_segments,
            query_vector,
            top_k=top_k * 3
        )

        # 2. 收集候选片段所在文件
        files_need_summary = set()
        for seg in candidate_segments:
            files_need_summary.add(seg.file_path)

        # 3. 对每个文件，计算摘要与query的相似度
        file_boost_factors = {}
        for file_path in files_need_summary:
            if file_path in self.summary_cache:
                summary = self.summary_cache[file_path]
                summary_sim = cosine_similarity(
                    query_vector,
                    self.encoder.encode(summary)
                )
                # 摘要相似度越高，该文件下的片段获得越高的提升
                file_boost_factors[file_path] = 1 + summary_sim

        # 4. 重新计算分数
        for seg in candidate_segments:
            boost = file_boost_factors.get(seg.file_path, 1.0)
            seg.boosted_score = seg.original_score * boost

        # 5. 排序返回
        candidate_segments.sort(key=lambda x: x.boosted_score, reverse=True)
        return candidate_segments[:top_k]
```

### 方案3：Query改写与片段扩展（推荐）

```python
class QueryExpansionWithSummary:
    """
    利用摘要增强查询，而非在结果层面做提升
    """

    def expand_query_with_summary(self, query, file_paths=None):
        """
        如果检索到相关的摘要，用摘要内容扩展query
        """
        # 1. 先用query检索摘要索引
        summary_results = self.summary_index.search(query, top_k=3)

        # 2. 如果摘要相关度高，用摘要内容扩展query
        if summary_results and summary_results[0].score > 0.7:
            top_summary = summary_results[0].content
            expanded_query = f"{query}\n\nContext from file overview:\n{top_summary}"
            return expanded_query

        return query

    def hybrid_search(self, query, top_k=10):
        """
        混合检索：摘要用于扩展，细节用于检索
        """
        # 扩展查询
        expanded_query = self.expand_query_with_summary(query)

        # 用扩展后的query检索细节
        results = self.detail_index.search(expanded_query, top_k=top_k)

        return results
```

## 三、高级优化技巧

### 1. **摘要质量保证**

```python
def ensure_summary_quality(summary, min_length=50, max_length=500):
    """
    确保摘要的质量：避免过短或过长
    """
    if len(summary) < min_length:
        # 补充结构信息
        summary += "\n[Note: File contains code implementation]"
    if len(summary) > max_length:
        # 截断但保留关键信息
        summary = summary[:max_length] + "..."
    return summary
```

### 2. **分层权重衰减**

```python
class HierarchicalWeighting:
    """
    层级权重：摘要匹配时，不同层级片段获得不同提升系数
    """
    boost_weights = {
        "class": 1.3,      # 类级别片段提升30%
        "function": 1.5,   # 函数级别提升50%
        "method": 1.4,     # 方法提升40%
        "block": 1.2       # 代码块提升20%
    }

    def apply_hierarchical_boost(self, results, matched_summaries):
        for result in results:
            entity_type = result.metadata.get("type", "block")
            base_boost = self.boost_weights.get(entity_type, 1.2)

            # 多个摘要匹配时累加提升
            total_boost = 1.0
            for summary in matched_summaries:
                if self.is_related(result, summary):
                    total_boost *= base_boost

            result.score *= total_boost
```

### 3. **去重与上下文融合**

```python
def deduplicate_and_merge(results):
    """
    去重并合并同一文件内的多个片段
    """
    grouped_by_file = {}
    for result in results:
        file_path = result.metadata["file_path"]
        if file_path not in grouped_by_file:
            grouped_by_file[file_path] = []
        grouped_by_file[file_path].append(result)

    final_results = []
    for file_path, segments in grouped_by_file.items():
        if len(segments) > 3:
            # 同一文件召回过多，合并展示
            merged = merge_segments(segments)
            final_results.append(merged)
        else:
            final_results.extend(segments)

    return final_results
```

## 四、实践经验与注意事项

| 场景                   | 推荐方案               | 原因                     |
| ---------------------- | ---------------------- | ------------------------ |
| 代码库较小 (<100文件)  | 方案2（联合向量+重排） | 实现简单，效果好         |
| 代码库较大 (>1000文件) | 方案1（双索引加权）    | 检索效率高，可扩展       |
| 需要精确控制相关性     | 方案3（Query扩展）     | 避免分数膨胀，结果更可控 |

**关键经验**：

1. **提升系数不宜过大**：建议1.2-1.5之间，过大会导致低质量片段因摘要匹配而被错误提升
2. **摘要本身不返回**：除非用户明确问"文件概览"，否则摘要只作为提升因子
3. **缓存摘要向量**：避免重复编码，提升检索效率
4. **A/B测试验证**：通过实际查询验证提升效果，避免过度依赖

这种设计本质上实现了一个**双路召回 + 交叉重排**的检索架构，既保持了片段的语义纯度，又利用了摘要的宏观信息来增强相关性判断。
